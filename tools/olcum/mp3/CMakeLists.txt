# W0-MP3a ölçüm prototipi (atılabilir; ürün koduna taşınmaz, 09 §2.1).
#   MP3_HEDEF=fw : kart aygıt yazılımı (C17/gnu17)
#   MP3_HEDEF=pc : PC aracı mp3_vcp ve masaüstü testleri mp3_test (C++20; ortak C kodu C olarak derlenir)
cmake_minimum_required(VERSION 3.20)

set(MP3_HEDEF "fw" CACHE STRING "fw ya da pc")
set(MP3_ORTAK ortak/mp3_cerceve.c ortak/mp3_kart.c ortak/mp3_halka.c)
set(MP3_UYARI_C -Wall -Wextra -Wconversion -Wshadow -Wvla -Wdouble-promotion)

if(MP3_HEDEF STREQUAL "pc")
  project(mp3_pc LANGUAGES C CXX)
  set(CMAKE_C_STANDARD 17)
  set(CMAKE_C_STANDARD_REQUIRED ON)
  set(CMAKE_C_EXTENSIONS OFF)
  set(CMAKE_CXX_STANDARD 20)
  set(CMAKE_CXX_STANDARD_REQUIRED ON)
  set(CMAKE_CXX_EXTENSIONS OFF)
  set(MP3_SANITIZE "" CACHE STRING "boş, asan (address+undefined), ubsan ya da tsan")
  set(_san "")
  if(MP3_SANITIZE STREQUAL "asan")
    set(_san -fsanitize=address,undefined -fno-omit-frame-pointer -fno-sanitize-recover=undefined)
  elseif(MP3_SANITIZE STREQUAL "ubsan")
    set(_san -fsanitize=undefined -fno-omit-frame-pointer -fno-sanitize-recover=undefined)
  elseif(MP3_SANITIZE STREQUAL "tsan")
    set(_san -fsanitize=thread -fno-omit-frame-pointer)
  endif()
  find_package(Threads REQUIRED)
  add_library(mp3_ortak STATIC ${MP3_ORTAK})
  target_include_directories(mp3_ortak PUBLIC ortak)
  target_compile_options(mp3_ortak PRIVATE -pedantic-errors ${MP3_UYARI_C} ${_san})
  add_library(mp3_pc_cekirdek STATIC pc/seri_posix.cpp pc/oturum.cpp pc/kayit.cpp)
  target_include_directories(mp3_pc_cekirdek PUBLIC pc)
  target_link_libraries(mp3_pc_cekirdek PUBLIC mp3_ortak Threads::Threads)
  add_executable(mp3_vcp pc/mp3_vcp.cpp)
  target_link_libraries(mp3_vcp PRIVATE mp3_pc_cekirdek)
  add_executable(mp3_test test/mp3_test.cpp test/sahte_hat.cpp)
  target_include_directories(mp3_test PRIVATE test)
  target_link_libraries(mp3_test PRIVATE mp3_pc_cekirdek)
  foreach(_t mp3_pc_cekirdek mp3_vcp mp3_test)
    target_compile_options(${_t} PRIVATE -Wall -Wextra -Wconversion -Wshadow -Wpedantic ${_san})
    target_link_options(${_t} PRIVATE ${_san})
  endforeach()
  return()
endif()

project(mp3_fw LANGUAGES C ASM)

if(NOT MP3_SRC)
  if(DEFINED ENV{IDA_CACHE} AND NOT "$ENV{IDA_CACHE}" STREQUAL "")
    set(MP3_SRC "$ENV{IDA_CACHE}/src")
  else()
    set(MP3_SRC "$ENV{HOME}/.cache/ida/src")
  endif()
endif()
set(MP3_SRC "${MP3_SRC}" CACHE PATH "ST kaynaklarının kökü")
set(MP3_KIMLIK "yok" CACHE STRING "derleme kimliği (derle.sh üretir; KIMLIK yanıtında döner)")

set(HAL_DIR  "${MP3_SRC}/stm32u5xx-hal-driver")
set(DEV_DIR  "${MP3_SRC}/cmsis-device-u5")
set(CORE_INC "${MP3_SRC}/STM32CubeU5/Drivers/CMSIS/Core/Include")
foreach(_d "${HAL_DIR}/Src" "${DEV_DIR}/Include" "${CORE_INC}")
  if(NOT EXISTS "${_d}")
    message(FATAL_ERROR "Kaynak yok: ${_d}  (tools/olcum/mp1/kaynak-indir.sh)")
  endif()
endforeach()

set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS ON)

set(MP3_HAL_MODULLER HAL CORTEX RCC GPIO PWR FLASH ICACHE)
file(READ "${HAL_DIR}/Inc/stm32u5xx_hal_conf_template.h" _conf)
string(REGEX REPLACE "#define (HAL_[A-Z0-9]+_MODULE_ENABLED)" "/* #define \\1 */" _conf "${_conf}")
string(REPLACE "/* #define HAL_MODULE_ENABLED */" "#define HAL_MODULE_ENABLED" _conf "${_conf}")
foreach(_m ${MP3_HAL_MODULLER})
  string(REPLACE "/* #define HAL_${_m}_MODULE_ENABLED */" "#define HAL_${_m}_MODULE_ENABLED" _conf "${_conf}")
endforeach()
file(WRITE "${CMAKE_BINARY_DIR}/uretilen/stm32u5xx_hal_conf.h.yeni" "${_conf}")
file(COPY_FILE "${CMAKE_BINARY_DIR}/uretilen/stm32u5xx_hal_conf.h.yeni"
               "${CMAKE_BINARY_DIR}/uretilen/stm32u5xx_hal_conf.h" ONLY_IF_DIFFERENT)

set(HAL_KAYNAK hal hal_cortex hal_rcc hal_rcc_ex hal_gpio hal_pwr hal_pwr_ex hal_flash hal_flash_ex hal_icache)
set(HAL_C "")
foreach(_s ${HAL_KAYNAK})
  list(APPEND HAL_C "${HAL_DIR}/Src/stm32u5xx_${_s}.c")
endforeach()

add_library(mp3_hal STATIC ${HAL_C} "${DEV_DIR}/Source/Templates/system_stm32u5xx.c")
target_include_directories(mp3_hal PUBLIC
  "${CMAKE_BINARY_DIR}/uretilen" "${HAL_DIR}/Inc" "${DEV_DIR}/Include" "${CORE_INC}")
target_compile_definitions(mp3_hal PUBLIC STM32U5A9xx USE_HAL_DRIVER)
target_compile_options(mp3_hal PRIVATE -w)

# Ortak test mantığı: ISO C17, uzantısız; HAL içerme yolu yok.
add_library(mp3_ortak STATIC ${MP3_ORTAK})
set_target_properties(mp3_ortak PROPERTIES C_EXTENSIONS OFF)
target_include_directories(mp3_ortak PUBLIC ortak)
target_compile_options(mp3_ortak PRIVATE -std=c17 -pedantic-errors ${MP3_UYARI_C} -fstack-usage)

set(MP3_LD "${DEV_DIR}/Source/Templates/gcc/linker/STM32U5A9xx_FLASH.ld")
set(MP3_STARTUP "${DEV_DIR}/Source/Templates/gcc/startup_stm32u5a9xx.s")

add_executable(mp3 fw/platform.c fw/kesmeler.c "${MP3_STARTUP}")
target_link_libraries(mp3 PRIVATE mp3_ortak mp3_hal)
target_compile_definitions(mp3 PRIVATE MP3_KIMLIK="${MP3_KIMLIK}")
target_compile_options(mp3 PRIVATE $<$<COMPILE_LANGUAGE:C>:-Wall -Wextra -Wconversion -Wdouble-promotion -fstack-usage>)
set_target_properties(mp3 PROPERTIES SUFFIX ".elf")
target_link_options(mp3 PRIVATE
  "-T${MP3_LD}" "-Wl,-Map=${CMAKE_BINARY_DIR}/mp3.map" "-Wl,--print-memory-usage"
  "-Wl,--wrap=malloc,--wrap=calloc,--wrap=realloc,--wrap=free")
add_custom_command(TARGET mp3 POST_BUILD
  COMMAND "${CMAKE_OBJCOPY}" -O binary "$<TARGET_FILE:mp3>" "${CMAKE_BINARY_DIR}/mp3.bin"
  COMMAND "${CMAKE_SIZE}" "$<TARGET_FILE:mp3>"
  VERBATIM)
